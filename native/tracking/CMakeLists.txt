cmake_minimum_required(VERSION 3.24)
project(cephvr_nvof LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Tracking's NVOF adapter requires 64-bit Windows")
endif()
find_package(CUDAToolkit REQUIRED)
find_path(NVOF_INCLUDE_DIR nvOpticalFlowCuda.h HINTS "${NVOF_SDK_ROOT}" PATH_SUFFIXES NvOFInterface REQUIRED)
add_library(cephvr_nvof SHARED nvof.cpp)
target_compile_features(cephvr_nvof PRIVATE cxx_std_17)
target_include_directories(cephvr_nvof PRIVATE "${NVOF_INCLUDE_DIR}")
target_link_libraries(cephvr_nvof PRIVATE CUDA::cuda_driver)
install(TARGETS cephvr_nvof RUNTIME DESTINATION cephvr/tracking/native)
